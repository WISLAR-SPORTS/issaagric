FROM dunglas/frankenphp:php8.3-bookworm

RUN install-php-extensions \
    intl \
    zip \
    gd \
    pdo_mysql \
    mbstring \
    exif \
    pcntl \
    bcmath

WORKDIR /app

COPY . .

RUN composer install --no-dev --optimize-autoloader

CMD ["php", "artisan", "serve", "--host=0.0.0.0", "--port=8000"]
